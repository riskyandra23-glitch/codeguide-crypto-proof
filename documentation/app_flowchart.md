flowchart TD
  Start[Start App]
  Start --> Connect[Connect Wallet Page]
  Connect --> FetchBalances[Call Balances API]
  FetchBalances --> Display[Display Aggregated Balances]
  Display --> Payment[Show Payment Button]
  Payment --> CheckoutAPI[Call CheckoutSession API]
  CheckoutAPI --> Stripe[Stripe Checkout]
  Stripe --> Webhook[Stripe Webhook]
  Webhook --> UpdateStatus[Update User Payment Status]
  UpdateStatus --> Success[Success Page]
  Success --> GenerateProof[Click Generate Proof]
  GenerateProof --> ProofAPI[Call Generate Proof API]
  ProofAPI --> CreatePDF[Generate PDF with QR Code]
  CreatePDF --> Download[Download PDF]
  Download --> Scan[Scan QR Code]
  Scan --> VerifyAPI[Call Verify Token API]
  VerifyAPI --> DisplayProof[Display Verified Proof Data]
